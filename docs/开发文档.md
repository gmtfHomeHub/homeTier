#### 启动网络实例

```rust
use easytier::launcher::NetworkConfig;
use easytier::instance_manager::NetworkInstanceManager;

pub async fn start_space_network(
    network_name: &str,
    network_secret: &str,
) -> Result<Uuid, String> {
    let manager = NetworkInstanceManager::new();

    let config = NetworkConfig::new()
        .network_name(network_name.to_string())
        .network_secret(network_secret.to_string())
        .dhcp(true)                          // 自动分配 IP
        .hostname(get_device_name())         // 设备名
        .build()
        .map_err(|e| e.to_string())?;

    let instance_id = manager
        .start_instance_task(config)
        .await
        .map_err(|e| e.to_string())?;

    Ok(instance_id)
}
```

### 3.3 配置管理

```rust
use easytier::launcher::NetworkConfig;
use easytier::common::config::TomlConfigLoader;

// 群配置 → TomlConfigLoader
fn group_config_to_loader(config: &GroupConfig) -> TomlConfigLoader {
    let cfg = TomlConfigLoader::default();
    cfg.set_network_identity(NetworkIdentity::new(
        &config.network_name,
        &config.network_secret,
    ));
    cfg.set_peers(config.peers.clone());
    cfg.set_listeners(config.listeners.clone());
    cfg.set_proxy_cidrs(config.proxy_cidrs.clone());
    cfg.set_acl(config.acl.clone());
    cfg
}

// 配置合并（群配置 > 本地配置 > 默认配置）
fn merge_and_build_config(
    group_cfg: Option<GroupConfig>,
    local_cfg: LocalConfig,
) -> NetworkConfig {
    let mut builder = NetworkConfig::new();

    // 群配置
    if let Some(g) = group_cfg {
        builder = builder
            .network_name(g.network_name)
            .network_secret(g.network_secret)
            .peers(g.peers)
            .listeners(g.listeners)
            .proxy_cidrs(g.proxy_cidrs);
    }

    // 本地配置（覆盖部分字段）
    builder = builder
        .hostname(local_cfg.hostname)
        .latency_first(local_cfg.latency_first)
        .mtu(local_cfg.mtu);

    builder.build().unwrap()
}
```

```typescript
// src/stores/spaceStore.ts
import { create } from 'zustand';
import { invoke } from '@tauri-apps/api/core';

interface Space {
  id: string;
  name: string;
  description?: string;
  network_name: string;
  members: Member[];
  status: 'disconnected' | 'connecting' | 'connected';
  virtual_ip?: string;
}

interface SpaceStore {
  spaces: Space[];
  currentSpaceId: string | null;
  loading: boolean;
  // Actions
  loadSpaces: () => Promise<void>;
  createSpace: (name: string, description?: string) => Promise<void>;
  joinSpace: (networkName: string, networkSecret: string) => Promise<void>;
  leaveSpace: (spaceId: string) => Promise<void>;
  setCurrentSpace: (spaceId: string | null) => void;
}

export const useSpaceStore = create<SpaceStore>((set, get) => ({
  spaces: [],
  currentSpaceId: null,
  loading: false,

  loadSpaces: async () => {
    set({ loading: true });
    try {
      const spaces = await invoke<Space[]>('list_spaces');
      set({ spaces, loading: false });
    } catch (e) {
      set({ loading: false });
      console.error('Failed to load spaces:', e);
    }
  },

  createSpace: async (name, description) => {
    const space = await invoke<Space>('create_space', { name, description });
    set((state) => ({ spaces: [...state.spaces, space] }));
  },

  joinSpace: async (networkName, networkSecret) => {
    const space = await invoke<Space>('join_space', {
      networkName,
      networkSecret,
    });
    set((state) => ({ spaces: [...state.spaces, space] }));
  },

  leaveSpace: async (spaceId) => {
    await invoke('leave_space', { spaceId });
    set((state) => ({
      spaces: state.spaces.filter((s) => s.id !== spaceId),
      currentSpaceId: state.currentSpaceId === spaceId ? null : state.currentSpaceId,
    }));
  },

  setCurrentSpace: (spaceId) => set({ currentSpaceId: spaceId }),
}));
```

### 5.3 Tauri 事件监听

```typescript
// src/hooks/useNetworkEvent.ts
import { useEffect } from 'react';
import { listen } from '@tauri-apps/api/event';
import { useSpaceStore } from '../stores/spaceStore';

interface NetworkStatusEvent {
  space_id: string;
  status: 'connected' | 'disconnected' | 'reconnecting';
  virtual_ip?: string;
  latency_ms?: number;
}

export function useNetworkEvents() {
  const updateSpaceStatus = useSpaceStore((s) => s.updateSpaceStatus);

  useEffect(() => {
    const unlisten = listen<NetworkStatusEvent>('network_status', (event) => {
      const { space_id, status, virtual_ip } = event.payload;
      updateSpaceStatus(space_id, status, virtual_ip);
    });

    return () => {
      unlisten.then((fn) => fn());
    };
  }, []);
}
```

## 6. 功能模块开发

### 6.1 空间管理模块

**实现步骤**：

1. 创建 `src-tauri/src/space/manager.rs` - 空间管理器（增删改查）
2. 创建 `src-tauri/src/space/store.rs` - 空间持久化存储
3. 创建 `src-tauri/src/commands/space.rs` - Tauri 命令处理器
4. 创建前端组件：`SpaceList`, `CreateSpaceDialog`, `JoinSpaceDialog`, `ShareSpaceDialog`

**核心流程**：

```
创建空间：
用户点击"创建空间" → 输入名称 → 前端调用 create_space
  → 后端生成 network-name (UUID) 和 network-secret
  → 启动 EasyTier 实例
  → 保存到 SQLite
  → 返回空间信息

加入空间：
用户点击链接/扫码 → 解析 network-name + network-secret
  → 前端调用 join_space
  → 后端启动 EasyTier 实例（使用解析的配置）
  → 等待 DHCP 分配 IP
  → 保存到 SQLite
  → 返回空间信息

分享空间：
用户点击"分享" → 后端生成分享链接 → 前端生成二维码
  → 链接格式: homeTier://join?name=xxx&secret=xxx
```

### 6.2 P2P 聊天模块

**实现步骤**：

1. 创建 `src-tauri/src/chat/server.rs` - HTTP 服务端（接收消息）
2. 创建 `src-tauri/src/chat/client.rs` - HTTP 客户端（发送消息）
3. 创建 `src-tauri/src/chat/message.rs` - 消息处理
4. 创建前端组件：`ChatView`, `MessageList`, `MessageInput`

**核心流程**：

```
启动聊天服务：
  空间连接成功 → 在虚拟 IP 上启动 HTTP 服务（随机端口）
  → 通过 EasyTier RPC 广播本节点地址（ip:port）
  → 接收其他节点的地址广播

发送消息：
  用户输入消息 → 前端调用 send_message
  → 后端构造消息对象（含签名）
  → 遍历空间内所有在线成员
  → 向每个成员的 HTTP 服务发送 POST 请求
  → 存储到本地 SQLite
  → 返回发送结果

接收消息：
  HTTP 服务收到 POST 请求 → 验证签名 → 存储到本地 SQLite
  → 通过 Tauri 事件发送到前端 → 前端显示
```

### 6.3 语音通话模块

**实现步骤**：

1. 创建 `src-tauri/src/voice/engine.rs` - WebRTC 音频引擎
2. 创建 `src-tauri/src/voice/signal.rs` - 信令交换
3. 创建前端组件：`VoicePanel`, `VoiceMember`, `VoiceControls`

**核心流程**：

```
加入语音频道：
  用户点击"加入语音" → 前端调用 join_voice_channel
  → 后端正交初始化 WebRTC PeerConnection
  → 通过 HTTP 信令交换 SDP Offer/Answer
  → 建立 P2P 音频连接
  → 每个成员维护与所有其他成员的连接

信令交换：
  WebRTC 需要信令通道来交换 SDP 和 ICE 候选
  使用 EasyTier 虚拟网络内的 HTTP 服务作为信令通道
  Offer → Answer → ICE Candidates 通过 HTTP 传输

音频处理：
  采集 → 编码 (Opus) → 发送 → 接收 → 解码 → 播放
  使用 VAD 检测说话状态
  混音器混合多个音频流
```

**WebRTC 依赖**：

```toml
# Cargo.toml
[dependencies]
webrtc = "0.11"           # WebRTC 库
tokio = { version = "1", features = ["full"] }
```

### 6.4 文件共享模块

**实现步骤**：

1. 创建 `src-tauri/src/file/transfer.rs` - 文件传输
2. 创建 `src-tauri/src/file/compress.rs` - Zstd 压缩
3. 创建 `src-tauri/src/file/crypto.rs` - 文件加密
4. 创建前端组件：`FileList`, `FileUpload`, `FileDownload`

**核心流程**：

```
发送文件：
  用户选择文件 → 前端调用 send_file
  → 后端读取文件
  → 如果启用压缩，使用 Zstd 压缩
  → 如果设置密码，使用 AES-256-GCM 加密
  → 在虚拟 IP 上启动临时 HTTP 文件服务器
  → 广播文件信息（文件名、大小、哈希、下载链接）
  → 接收方通过 HTTP 下载

接收文件：
  收到文件广播 → 前端显示文件信息 → 用户点击下载
  → 前端调用 receive_file
  → 后端通过 HTTP 下载（支持断点续传）
  → 如果加密，使用密码解密
  → 如果压缩，解压
  → 保存到本地
  → 校验文件完整性（SHA256）
  → 显示传输进度
```

**核心流程**：

```
发起共享：
  用户点击"屏幕共享" → 选择屏幕/窗口
  → 前端调用 start_screen_share
  → 后端使用平台 API 采集屏幕画面
  → 通过 WebRTC 视频轨传输
  → 被邀请者接收并显示

采集方案：

```

## 9. 常见问题

### 9.1 Windows 虚拟网卡驱动

**问题**：Windows 下需要管理员权限安装 wintun 驱动。

**解决方案**：
- 打包时将 wintun.dll 放在 `src-tauri/resources/` 目录
- 应用启动时检测是否有管理员权限
- 没有权限时弹出 UAC 提权对话框
- 提权后自动安装虚拟网卡

### 9.2 macOS 权限问题

**问题**：macOS 下创建 utun 设备需要 root 权限。

**解决方案**：
- 使用 `AuthorizationExecuteWithPrivileges` 提权
- 创建 helper 工具（SMJobBless）实现无密码提权
- 或者在首次启动时提示用户输入管理员密码

### 9.3 Android VPN 授权

**问题**：Android 需要用户手动授权 VPN 连接。

**解决方案**：
- 使用 `VpnService.prepare()` 触发授权 Intent
- 授权后保持前台服务（Foreground Service）防止被杀
- 使用通知栏常驻提醒用户

### 9.4 iOS NetworkExtension

**问题**：iOS 需要配置 NetworkExtension entitlement。

**解决方案**：
- 在 Apple Developer Center 启用 Network Extension 能力
- 在 Xcode 中配置 Packet Tunnel Provider
- 打包时使用正确的 provisioning profile

### 9.5 移动端后台保活

**问题**：移动端应用进入后台后可能被系统杀死。

**解决方案**：
- **Android**：使用 Foreground Service + 通知栏常驻
- **iOS**：通过 VPN 配置保持后台连接（系统自动管理）
- 在合适的时机重新连接
